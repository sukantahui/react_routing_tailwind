2. COMMON MODULE WORKBOOK & SHEET ARCHITECTURE
--------------------------------------------------------------------------------
• A SINGLE COMMON `.xlsx` WORKBOOK per Module:
    - Every module has ONE central, comprehensive Excel workbook stored in:
        `./excel_files/[module_name].xlsx`
    - This single workbook serves as the master interactive worksheet for all topics in that chapter.
    - Inside this workbook, the FIRST sheet (`SheetNames[0]`) is ALWAYS the executive **Overview** landing sheet.
    - Each subsequent topic has its own dedicated worksheet (e.g. `Topic0_Spill_Engine`, `Topic1_Calc`, `VLOOKUP_Table`, etc.).
    - When a topic requires multi-table comparisons or relational joins, multiple related sheets can be included (e.g. `Orders_Data`, `Customers_Master`).

    - Every `Topic[N].jsx` must include an intuitive, high-visibility **Download Practice Workbook (.xlsx)** button.
    - The button downloads the common module `.xlsx` file directly to the learner's computer so they can open and practice in their local Microsoft Excel / Office software.
    - Uses Vite's `?url` asset import to obtain the direct downloadable asset URL:
        import sampleWorkbookUrl from "./excel_files/[module_name].xlsx?url";

      cell.font = { name: 'Segoe UI', size: 10, bold: true, color: { argb: 'FFFFFFFF' } };
      cell.fill = { type: 'pattern', pattern: 'solid', fgColor: { argb: themeColor } };
      cell.alignment = { vertical: 'middle', horizontal: 'center' };
      cell.border = {
        top: { style: 'medium', color: { argb: 'FF0F172A' } },

      };
    });

    // Populate Data Rows (Zebra striping & thin borders)
    dataRows.forEach((rowObj, rIdx) => {
      const rowNum = rIdx + 3;
      const row = ws.getRow(rowNum);
      row.height = 20;
      const isEven = rIdx % 2 === 0;

      columns.forEach((col, cIdx) => {
        const colLetter = String.fromCharCode(65 + cIdx);
        const cell = ws.getCell(`${colLetter}${rowNum}`);
        cell.value = rowObj[col.key];
        cell.font = { name: 'Segoe UI', size: 9.5, color: { argb: 'FF0F172A' } };
        cell.alignment = { vertical: 'middle', horizontal: col.alignment || 'left', indent: col.alignment === 'center' ? 0 : 1 };
        cell.fill = { type: 'pattern', pattern: 'solid', fgColor: { argb: isEven ? 'FFFFFFFF' : 'FFF8FAFC' } };
        cell.border = {
          top: { style: 'thin', color: { argb: 'FFE2E8F0' } },
          bottom: { style: 'thin', color: { argb: 'FFE2E8F0' } },
          left: { style: 'thin', color: { argb: 'FFE2E8F0' } },
          right: { style: 'thin', color: { argb: 'FFE2E8F0' } }
        };
      });
    });

    return ws;
  }

  // Create child sheets for each topic
  // e.g. addStyledTopicSheet('Topic0_Overview', 'FF0F172A', sampleColumns, sampleData);

  // Ensure output directory exists
  const outputDir = path.join(__dirname, 'excel_files');
  if (!fs.existsSync(outputDir)) fs.mkdirSync(outputDir, { recursive: true });

  const workbookPath = path.join(outputDir, 'module_workbook.xlsx');
  await wb.xlsx.writeFile(workbookPath);
  console.log(`✓ Master module workbook successfully generated at: ${workbookPath}`);
}
