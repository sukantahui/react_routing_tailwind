2. WORKSHOP CHALLENGE 1: THE 360° ACADEMIC PERFORMANCE SCORECARD
--------------------------------------------------------------------------------
Combines department ranking, academy percentile ranking, valedictorian gap, 
and trailing moving averages into a single high-performance analytical query:

    WITH AcademicMetrics AS (
        SELECT 
            s.student_id,
            CONCAT(s.first_name, ' ', s.last_name) AS student_name,
            d.dept_name,
            s.exam_score_pct,
            -- 1. Department Rank:
            DENSE_RANK() OVER w_dept AS dept_rank,
            -- 2. Academy Percentile:
            ROUND(PERCENT_RANK() OVER w_global * 100.0, 1) AS academy_percentile,
            -- 3. Valedictorian Gap:
            ROUND(FIRST_VALUE(s.exam_score_pct) OVER (
                PARTITION BY s.dept_id ORDER BY s.exam_score_pct DESC
            ) - s.exam_score_pct, 2) AS gap_to_topper,
            -- 4. Trailing 3-Quiz Moving Average:
            ROUND(AVG(s.exam_score_pct) OVER (
                PARTITION BY s.student_id ORDER BY s.created_at ROWS 2 PRECEDING
            ), 2) AS trailing_3quiz_avg
        FROM students s
        JOIN departments d ON s.dept_id = d.dept_id
        WINDOW 
            w_dept AS (PARTITION BY s.dept_id ORDER BY s.exam_score_pct DESC),
            w_global AS (ORDER BY s.exam_score_pct ASC)
    )
    SELECT * FROM AcademicMetrics ORDER BY dept_name, dept_rank;
