1. FORMAT SPECIFIERS:
   - '%s' : String
   - '%d' : Integer
   - '%f' : Float / Double (e.g. '%.2f')
   - '%b' : Boolean
   - '%x' : Hexadecimal
   - '%n' : Platform newline
