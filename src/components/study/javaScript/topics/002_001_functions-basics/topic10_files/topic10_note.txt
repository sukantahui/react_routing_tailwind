3. EXCEPTIONS, CORNER CASES, COERCIONS & QUIRKS
--------------------------------------------------------------------------------
[!] fn.length Desync with Default Parameters: Default and rest parameters are NOT counted in `fn.length`, causing naive curry functions to terminate too early.
[!] Stack Overflow on Infinite Currying: Infinite currying without value serialization or terminal invocations exhausts memory.
[!] Loss of this in Curried Chains: Chained nested functions default their `this` context to undefined/global unless bound.

Example 5: Senior Hidden Gem Pattern
function curry(fn) {
  return function curried(...args) {
    if (args.length >= fn.length) {
      return fn.apply(this, args);
    }
    return function(...nextArgs) {
      return curried.apply(this, args.concat(nextArgs));
    };
  };
}
const multiply3 = (a, b, c) => a * b * c;
const curriedMult = curry(multiply3);
console.log(curriedMult(2)(3)(4)); // 24
console.log(curriedMult(2, 3)(4)); // 24
