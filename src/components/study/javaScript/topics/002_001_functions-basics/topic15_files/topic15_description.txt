4. GROUNDED PRACTICAL CODE EXAMPLES (PROGRESSIVE LEARNING)
--------------------------------------------------------------------------------
--- Example 1: Memoized Computation Engine via Closure ---
function createMemoizedCalculator() {
  const cache = {}; // Private cache in closure
  
  return function(num) {
    if (num in cache) {
      console.log('Fetching from cache for input ' + num + '...');
      return cache[num];
    }
    console.log('Computing expensive square for input ' + num + '...');
    const result = num * num;
    cache[num] = result;
    return result;
  };
}

const calcSquare = createMemoizedCalculator();
console.log(calcSquare(25)); // Computes: 625
console.log(calcSquare(25)); // Cached: 625 (Instant!)

Teacher's Note on Example 1:
Uses closures to preserve a private memoization cache across invocations.
