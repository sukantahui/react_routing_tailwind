--- Example 1: Enterprise Banking System with Encapsulation and Inheritance ---
class BankAccount {
  #balance;
  #accountNumber;
  
  constructor(accNum, initialDeposit) {
    this.#accountNumber = accNum;
    this.#balance = initialDeposit;
  }
  
  deposit(amt) {
    if (amt <= 0) throw new Error('Deposit must be positive');
    this.#balance += amt;
    return `Deposited ₹${amt}. New Balance: ₹${this.#balance}`;
  }
  
  withdraw(amt) {
    if (amt > this.#balance) return 'Insufficient funds!';
    this.#balance -= amt;
    return `Withdrew ₹${amt}. Remaining: ₹${this.#balance}`;
  }
  
  getBalance() { return this.#balance; }
}

class SavingsAccount extends BankAccount {
  constructor(accNum, deposit, interestRate = 0.05) {
    super(accNum, deposit);
    this.interestRate = interestRate;
  }
  
  applyInterest() {
    const interest = this.getBalance() * this.interestRate;
    this.deposit(interest);
    return `Interest of ₹${interest.toFixed(2)} applied!`;
  }
}

const savings = new SavingsAccount('SB-1009', 10000);
console.log(savings.deposit(2000));
console.log(savings.applyInterest());
console.log('Final Balance: ₹' + savings.getBalance());

Teacher's Note on Example 1:
Combines private fields, inheritance, super, and polymorphic domain behavior into a unified OOP architecture.
