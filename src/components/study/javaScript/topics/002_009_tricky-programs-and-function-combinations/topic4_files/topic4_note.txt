4. 💎 SPECIAL LANGUAGE FEATURES & SENIOR PRO SECRETS
--------------------------------------------------------------------------------
- Secret 1: High-Performance Algorithmic Composition
  // 💎 SENIOR SECRET: High-Performance O(1) LRU Memoizer
function memoizeLRU(fn, capacity = 100, keyResolver = (...args) => args.join("|")) {
  const cache = new Map();
  return function(...args) {
    const key = keyResolver(...args);
    if (cache.has(key)) {
      const val = cache.get(key);
      cache.delete(key); // Refresh key position for LRU
      cache.set(key, val);
      return val;
    }
    const res = fn.apply(this, args);
    if (cache.size >= capacity) {
      const oldestKey = cache.keys().next().value;
      cache.delete(oldestKey);
    }
    cache.set(key, res);
    return res;
  };
}
