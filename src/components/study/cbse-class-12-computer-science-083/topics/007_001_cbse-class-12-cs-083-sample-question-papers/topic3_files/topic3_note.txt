# Mock Paper 2 - Binary File Deletion Strategy (5 Marks)
import pickle, os

def remove_inactive(roll):
    with open('student.dat', 'rb') as fin, open('temp.dat', 'wb') as fout:
        try:
            while True:
                rec = pickle.load(fin)
                if rec[0] != roll:
                    pickle.dump(rec, fout)
        except EOFError:
            pass
    os.remove('student.dat')
    os.rename('temp.dat', 'student.dat')
