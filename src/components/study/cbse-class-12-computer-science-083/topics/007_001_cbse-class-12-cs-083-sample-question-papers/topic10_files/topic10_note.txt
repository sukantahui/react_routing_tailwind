# Binary File In-Place Update (5 Marks)
import pickle

def update_marks(roll, new_marks):
    with open('student.dat', 'rb+') as f:
        try:
            while True:
                pos = f.tell()
                rec = pickle.load(f)
                if rec[0] == roll:
                    rec[2] = new_marks
                    f.seek(pos)
                    pickle.dump(rec, f)
                    break
        except EOFError:
            pass
