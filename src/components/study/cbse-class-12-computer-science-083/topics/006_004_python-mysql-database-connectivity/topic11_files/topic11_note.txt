3. SYNTAX & CODE SNIPPET:
import mysql.connector

def get_connection():
    return mysql.connector.connect(host="localhost", user="root", password="tiger", database="SchoolDB")

def insert_student():
    con = get_connection()
    cur = con.cursor()
    r = int(input("Roll No: "))
    n = input("Name: ")
    m = float(input("Marks: "))
    cur.execute("INSERT INTO Student VALUES (%s, %s, %s);", (r, n, m))
    con.commit()
    print("Student enrolled successfully!")
    cur.close()
    con.close()

def view_all():
    con = get_connection()
    cur = con.cursor()
    cur.execute("SELECT * FROM Student;")
    rows = cur.fetchall()
    for row in rows:
        print(row)
    cur.close()
    con.close()

4. CBSE COMMON PITFALLS:
Always remember to commit transactions and close both cursor and connection.
