# Practice Challenge: Complete the function to display employees in Dept 'IT'
import mysql.connector

def display_it_employees():
    con = mysql.connector.connect(host="localhost", user="root", password="tiger", database="CompanyDB")
    cur = con.cursor()
    # Write the execute statement and loop here:
    cur.execute("SELECT EmpId, EName, Salary FROM Employee WHERE Dept = 'IT';")
    for row in cur.fetchall():
        print(row[0], row[1], row[2])
    cur.close()
    con.close()
