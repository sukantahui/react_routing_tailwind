2. FULL RECIPE TO RENDER PAIRPLOT:
--------------------------------------------------------------------------------
import seaborn as sns
import matplotlib.pyplot as plt

sns.set_theme(style="ticks", palette="Set1")

g = sns.pairplot(
    data=df,
    hue="Track",
    diag_kind="kde",
    diag_kws={"fill": True, "alpha": 0.4},
    plot_kws={"s": 70, "alpha": 0.85, "edgecolor": "white"}
)

# Adjust title position for Figure-level objects
g.fig.subplots_adjust(top=0.92)
g.fig.suptitle("Barrackpore Batch: Multivariable Feature Exploration", fontsize=12)
