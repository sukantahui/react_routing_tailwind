1. BASIC MULTI-ARGUMENT PARAMETRIZATION
--------------------------------------------------------------------------------
  import pytest

  @pytest.mark.parametrize("score, expected_grade", [
      (95.0, "A+"),
      (90.0, "A+"), # Boundary check
      (80.0, "A"),
      (60.0, "B"),
      (40.0, "C"),
      (35.0, "F"),
  ])
  def test_grade_boundaries(score, expected_grade):
      # PyTest generates 6 individual independent test cases!
      assert calculate_grade(score) == expected_grade

      pytest.param(100, 90, id="regular_10pct_waiver"),
      pytest.param(0, 0, id="zero_fee_edge_case"),
      pytest.param(-50, None, marks=pytest.mark.xfail, id="known_negative_bug"),
  ])

      assert apply_waiver(input_val) == expected
