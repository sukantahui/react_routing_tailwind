2. MODULAR FIXTURES & DEPENDENCY INJECTION
--------------------------------------------------------------------------------
  import pytest

  @pytest.fixture(scope="module")
  def database_conn():
      # Setup phase
      conn = init_in_memory_db()
      yield conn # Hands over fixture to test
      # Teardown phase (runs automatically after tests in module finish)
      conn.close()

  @pytest.fixture
  def fresh_student(database_conn):
      # Fixture requesting another fixture!
      student = Student("STU_1", "Mamata", "Barrackpore", 10000.0)
      database_conn.save(student)
      return student

  def test_student_payment(fresh_student):
      # Dependency injection via function parameter name
      assert fresh_student.name == "Mamata"
      fresh_student.pay(3000.0)
      assert fresh_student.balance == 7000.0

3. PYTEST CLI POWER COMMANDS
--------------------------------------------------------------------------------
  pytest                      # Auto-discovers and runs all tests
  pytest -v                   # Verbose output
  pytest -k "scholarship"     # Runs only tests matching name substring
  pytest -x                   # Stop immediately on first test failure
  pytest --lf                 # Re-run only failed tests from last run
