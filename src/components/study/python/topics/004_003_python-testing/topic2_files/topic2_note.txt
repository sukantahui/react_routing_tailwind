  import unittest

  class TestStudentModule(unittest.TestCase):

      def test_admission_discount(self):
          # Specialized TestCase assertions
          self.assertEqual(calculate_fee(10000.0, 0.15), 8500.0)
          self.assertTrue(is_eligible(95.0))
          self.assertIn("PYTHON", ["PYTHON", "SQL", "WEB"])

2. FIXTURE LIFECYCLE HOOKS
--------------------------------------------------------------------------------
  class TestDatabaseIntegration(unittest.TestCase):

      @classmethod
      def setUpClass(cls):
          # Runs ONCE before all tests in this class (e.g. create DB connection)
          cls.db = init_in_memory_db()

      @classmethod
      def tearDownClass(cls):
          # Runs ONCE after all tests in this class (e.g. close DB connection)
          cls.db.close()

      def setUp(self):
          # Runs BEFORE EACH test method (e.g. clear tables, setup fresh test data)
          self.db.clear_tables()

      def tearDown(self):
          # Runs AFTER EACH test method (e.g. rollback transactions)
          self.db.rollback()

--------------------------------------------------------------------------------
3. RUNNING FROM CLI
--------------------------------------------------------------------------------
  python -m unittest discover -s tests -p "test_*.py" -v
