1. THE UNIVERSAL DECORATOR TEMPLATE
--------------------------------------------------------------------------------
  import functools

  def my_decorator(func):
      @functools.wraps(func)  # Mandatory to preserve __name__ and __doc__!
      def wrapper(*args, **kwargs):
          # 1. Pre-execution logic
          print("Before function execution")

          # 2. Execute original function
          result = func(*args, **kwargs)

          # 3. Post-execution logic
          print("After function execution")

          # 4. Mandatory return!
          return result
      return wrapper

--------------------------------------------------------------------------------
2. THE `@` SYNTACTIC SUGAR EQUIVALENCE
--------------------------------------------------------------------------------
  @my_decorator
  def process():
      pass

  # Identical to:
  # def process(): pass
  # process = my_decorator(process)
