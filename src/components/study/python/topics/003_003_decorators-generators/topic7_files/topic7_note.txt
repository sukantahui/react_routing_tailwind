1. CUSTOM ITERATOR CLASS SKELETON
--------------------------------------------------------------------------------
  class MyIterator:
      def __init__(self, data):
          self.data = data
          self.cursor = 0

      def __iter__(self):
          return self  # Idempotence: returns self

      def __next__(self):
          if self.cursor >= len(self.data):
              raise StopIteration("Stream completed")
          item = self.data[self.cursor]
          self.cursor += 1
          return item
