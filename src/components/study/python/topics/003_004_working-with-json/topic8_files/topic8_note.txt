1. ATOMIC WRITE PATTERN (ZERO 0-BYTE CORRUPTIONS)
--------------------------------------------------------------------------------
  def atomic_save(data, target_path):
      target_dir = os.path.dirname(os.path.abspath(target_path))
      fd, tmp_path = tempfile.mkstemp(dir=target_dir, prefix="db_tmp_", suffix=".json")
      with os.fdopen(fd, "w", encoding="utf-8") as f:
          json.dump(data, f, indent=2)
          f.flush()
          os.fsync(f.fileno())  # Commit buffers to physical disk
      os.replace(tmp_path, target_path)  # Atomic filesystem inode swap
