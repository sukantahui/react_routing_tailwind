1. SAFE PATH TRAVERSAL PATTERN
--------------------------------------------------------------------------------
  def safe_get(data, path, default=None):
      curr = data
      for token in path:
          if isinstance(curr, dict) and isinstance(token, str):
              curr = curr.get(token)
          elif isinstance(curr, (list, tuple)) and isinstance(token, int):
              curr = curr[token] if 0 <= token < len(curr) else None
          else: return default
          if curr is None: return default
      return curr

  def redact_pii(data):
      if isinstance(data, dict):
          return {k: "********" if "password" in k or "token" in k else redact_pii(v) for k, v in data.items()}
      elif isinstance(data, list):
          return [redact_pii(x) for x in data]
      return data
