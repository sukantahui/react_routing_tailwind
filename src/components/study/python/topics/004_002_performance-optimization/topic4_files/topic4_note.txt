1. __SLOTS__ SYNTAX & BEST PRACTICES
--------------------------------------------------------------------------------
  # Explicit Class with __slots__:
  class StudentRecord:
      __slots__ = ("student_id", "name", "campus", "fee")

      def __init__(self, student_id, name, campus, fee):
          self.student_id = student_id
          self.name = name
          self.campus = campus
          self.fee = fee

  # Modern Python 3.10+ Dataclass with slots=True:
  from dataclasses import dataclass

  @dataclass(slots=True)
  class StudentDossier:
      student_id: str
      name: str
      campus: str
      fee: float

2. MEMORY PROFILING WITH TRACEMALLOC
--------------------------------------------------------------------------------
  import tracemalloc

  tracemalloc.start()
  snapshot1 = tracemalloc.take_snapshot()

  # Execute memory-heavy workload:
  objects = [StudentRecord(...) for _ in range(50000)]

  snapshot2 = tracemalloc.take_snapshot()
  curr, peak = tracemalloc.get_traced_memory()
  tracemalloc.stop()

  # View top allocating lines:
  top_stats = snapshot2.compare_to(snapshot1, 'lineno')
  for stat in top_stats[:5]:
      print(stat)
