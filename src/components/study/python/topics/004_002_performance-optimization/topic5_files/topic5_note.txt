  result = []

  # BETTER: Local method caching (Fast C bytecode resolution)
  result = []
  append = result.append

2. ELIMINATING QUADRATIC NESTED SEARCHES VIA HASH INDEXING
--------------------------------------------------------------------------------
  # SLOW: O(N * M) Quadratic Nested Search
  common_students = []
  for stu in batch_a:
      if stu["id"] in [b["id"] for b in batch_b]: # Rebuilds list on every step!
          common_students.append(stu)

  # FAST: O(N + M) Linear Set Hash Indexing
  batch_b_ids = {b["id"] for b in batch_b} # Built once in O(M)
  common_students = [stu for stu in batch_a if stu["id"] in batch_b_ids] # O(1) lookups!
