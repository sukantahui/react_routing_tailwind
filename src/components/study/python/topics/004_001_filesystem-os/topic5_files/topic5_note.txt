1. SUBPROCESS.RUN() (SYNCHRONOUS)
--------------------------------------------------------------------------------
  res = subprocess.run(
      ["git", "status", "--porcelain"],
      capture_output=True,
      text=True,
      check=True,
      timeout=5.0
  )
  * res.returncode : Exit code (0 = success)
  * res.stdout     : Decoded string output

--------------------------------------------------------------------------------
2. SUBPROCESS.POPEN() (STREAMING & ASYNC)
--------------------------------------------------------------------------------
  proc = subprocess.Popen(
      ["python", "long_worker.py"],
      stdout=subprocess.PIPE,
      text=True
  )
  for line in proc.stdout:
      print("Stream:", line.strip())
  proc.wait()

--------------------------------------------------------------------------------
3. PIPELINE CHAINING (p1 | p2)
--------------------------------------------------------------------------------
  p1 = subprocess.Popen(["cat", "data.csv"], stdout=subprocess.PIPE)
  p2 = subprocess.Popen(["grep", "PAID"], stdin=p1.stdout, stdout=subprocess.PIPE)
  p1.stdout.close() # Important pipe closing invariant!
