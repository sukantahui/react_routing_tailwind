1. CAPSTONE ARCHITECTURE
   The suite unifies five core metaprogramming concepts:
   1. Multi-tier Variadic Logging with Level Filtering (LOG_D, LOG_I, LOG_W, LOG_E)
   2. Source Location Injection (__FILE__, __LINE__, __func__)
   3. Custom Panic Assertions with Stringized Expressions (#condition)
   4. High-Resolution Performance Benchmarking (TIME_BLOCK)
   5. Type-Generic Array Capacity Extraction (ARRAY_SIZE)

2. INDUSTRIAL VARIADIC LOGGING PATTERN
   #define LOG_DISPATCH(lvl_str, lvl, fmt, ...) \
       do { \
           if ((lvl) >= ACTIVE_LOG_LEVEL) { \
               fprintf(stdout, "[%-5s] [%s:%d in %s()] " fmt "\n", \
                       lvl_str, __FILE__, __LINE__, __func__, ##__VA_ARGS__); \
           } \
       } while (0)

3. CUSTOM PANIC ASSERTION
   #define PANIC_ASSERT(cond, msg) \
       do { \
           if (!(cond)) { \
               fprintf(stderr, "[PANIC] Failed: '%s' | Msg: %s | %s:%d\n", \
                       #cond, msg, __FILE__, __LINE__); \
               abort(); \
           } \
       } while (0)

4. PERFORMANCE TIMING BLOCK
   #define TIME_BLOCK(name, code) \
       do { \
           clock_t s_##name = clock(); \
           code \
           clock_t e_##name = clock(); \
           double ms = ((double)(e_##name - s_##name) / CLOCKS_PER_SEC) * 1000.0; \
           printf("[BENCHMARK] '%s': %.4f ms\n", #name, ms); \
       } while (0)

5. ARRAY_SIZE MACRO
   #define ARRAY_SIZE(arr) (sizeof(arr) / sizeof((arr)[0]))
   Note: Only valid on local stack/global arrays, NOT decayed pointer parameters!
